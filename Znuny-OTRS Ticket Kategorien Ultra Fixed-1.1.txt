        localStorage.setItem(STORAGE_KEY, JSON.stringify(state));

    function hasManualCategory(ticketId) {
        return Object.prototype.hasOwnProperty.call(state.categories, ticketId);
    }

    function getManualCategory(ticketId) {
        return hasManualCategory(ticketId) ? state.categories[ticketId] : null;
    }

    function setManualCategory(ticketId, category) {
        if (category === "__auto") {
            delete state.categories[ticketId];
        } else {
            state.categories[ticketId] = category;
        }

        save();
    }

    function parseAge(text) {
        return ((text.match(/(\d+)\s*d/)?.[1] || 0) * 1440)
            + ((text.match(/(\d+)\s*h/)?.[1] || 0) * 60)
            + Number(text.match(/(\d+)\s*m/)?.[1] || 0);
    }

    function findTable() {
        return [...document.querySelectorAll("table")].find(table =>
            table.innerText.includes("CASE") &&
            table.innerText.includes("ALTER") &&
            table.innerText.includes("SENDER") &&
            table.innerText.includes("TITEL")
        );
    }

    function getIndexes(table) {
        const headers = [...table.querySelectorAll("th")].map(th => th.innerText.trim().toUpperCase());

        return {
            case: headers.findIndex(h => h.includes("CASE")),
            age: headers.findIndex(h => h.includes("ALTER")),
            sender: headers.findIndex(h => h.includes("SENDER")),
            title: headers.findIndex(h => h.includes("TITEL")),
            status: headers.findIndex(h => h.includes("STATUS")),
            customer: headers.findIndex(h => h.includes("KUNDENNUMMER"))
        };
    }

    function getCellText(row, index) {
        if (index < 0) return "";
        return row.querySelectorAll("td")[index]?.innerText.trim() || "";
    }

    function getTicketId(row, indexes) {
        const cell = row.querySelectorAll("td")[indexes.case];
        if (!cell) return "";

        if (cell.dataset.ticketId) return cell.dataset.ticketId;

        const id = cell.innerText.trim().match(/\b\d{7,}\b/)?.[0] || "";
        if (id) cell.dataset.ticketId = id;

        return id;
    }

    function getSearchText(row, indexes) {
        return [
            getCellText(row, indexes.case),
            getCellText(row, indexes.sender),
            getCellText(row, indexes.title),
            getCellText(row, indexes.status),
            getCellText(row, indexes.customer)
        ].join(" ").toLowerCase();
    }

    function getAgeMinutes(row, indexes) {
        return parseAge(getCellText(row, indexes.age));
    }



 function autoDetectCategory(row, indexes) {
    const text = getSearchText(row, indexes);
    const age = getAgeMinutes(row, indexes);

    if (age >= 100 * 1440) {
        return "langzeit";
    }

    const priority = [
        "erledigt",
        "warten",
        "dringend",
        "studis",
        "hardware",
        "software",
        "langzeit"
    ];

    for (const groupId of priority) {
        const words = KEYWORDS[groupId] || [];
        if (words.some(word => text.includes(word.toLowerCase()))) {
            return groupId;
        }
    }

    return "";
}

   function getEffectiveCategory(row, indexes) {
    const ticketId = getTicketId(row, indexes);
    const manual = getManualCategory(ticketId);

    if (manual !== null) {
        return manual;
    }

    return autoDetectCategory(row, indexes);
}

    function isStarred(row, indexes) {
        const cells = [...row.querySelectorAll("td")];
        const beforeCase = cells.slice(0, indexes.case);

        return beforeCase.some(cell => {
            const text = cell.innerText.trim();
            const html = cell.innerHTML.toLowerCase();

            return text.includes("★") ||
                html.includes("star") ||
                html.includes("important") ||
                html.includes("flag") ||
                html.includes("priority");
        });
    }

    function buildToolbar(table) {
        if (document.querySelector("#tm-toolbar")) return;

        const toolbar = document.createElement("div");
        toolbar.id = "tm-toolbar";
        toolbar.className = "tm-toolbar";
        toolbar.innerHTML = `<strong>Kategorien:</strong>`;

        const buttons = [
            { id: "all", label: "Alle" },
            ...GROUPS.map(g => ({ id: g.id || "none", label: g.short }))
        ];

        buttons.forEach(btn => {
            const button = document.createElement("button");
            button.type = "button";
            button.className = "tm-filter-btn";
            button.textContent = btn.label;
            button.dataset.filter = btn.id;

            button.addEventListener("click", e => {
                e.preventDefault();
                e.stopPropagation();

                activeFilter = btn.id;

                document.querySelectorAll(".tm-filter-btn").forEach(b => b.classList.remove("active"));
                button.classList.add("active");

                apply();
            });

            if (btn.id === activeFilter) {
                button.classList.add("active");
            }

            toolbar.appendChild(button);
        });

        table.parentElement.insertBefore(toolbar, table);
    }

    function updateToolbarCounts(counts) {
        document.querySelectorAll(".tm-filter-btn").forEach(button => {
            const filter = button.dataset.filter;

            if (filter === "all") {
                const total = Object.values(counts).reduce((a, b) => a + b, 0);
                button.textContent = `Alle (${total})`;
                return;
            }

            if (filter === "none") {
                button.textContent = `Keine (${counts[""] || 0})`;
                return;
            }

            const group = GROUPS.find(g => g.id === filter);
            if (group) {
                button.textContent = `${group.short} (${counts[group.id] || 0})`;
            }
        });
    }

    function blockTicketNavigation(element) {
        ["click", "mousedown", "mouseup", "pointerdown", "pointerup", "dblclick"].forEach(ev => {
            element.addEventListener(ev, e => {
                e.stopPropagation();
            });
        });
    }

    function closeAllNotePopups() {
        document.querySelectorAll(".tm-note-popup").forEach(p => p.remove());
        openNoteTicketId = null;
    }

    function openNotePopup(button, ticketId) {
        closeAllNotePopups();

        openNoteTicketId = ticketId;

        const rect = button.getBoundingClientRect();

        const popup = document.createElement("div");
        popup.className = "tm-note-popup";
        popup.style.left = `${Math.min(rect.left, window.innerWidth - 260)}px`;
        popup.style.top = `${rect.bottom + 6}px`;

        const title = document.createElement("div");
        title.className = "tm-note-popup-title";
        title.textContent = `Notiz für Ticket ${ticketId}`;

        const textarea = document.createElement("textarea");
        textarea.placeholder = "Lokale Notiz...";
        textarea.value = state.notes[ticketId] || "";

        const actions = document.createElement("div");
        actions.className = "tm-note-actions";

        const deleteBtn = document.createElement("button");
        deleteBtn.type = "button";
        deleteBtn.textContent = "Löschen";

        const cancelBtn = document.createElement("button");
        cancelBtn.type = "button";
        cancelBtn.textContent = "Abbrechen";

        const saveBtn = document.createElement("button");
        saveBtn.type = "button";
        saveBtn.textContent = "Speichern";

        actions.appendChild(deleteBtn);
        actions.appendChild(cancelBtn);
        actions.appendChild(saveBtn);

        popup.appendChild(title);
        popup.appendChild(textarea);
        popup.appendChild(actions);

        blockTicketNavigation(popup);
        blockTicketNavigation(textarea);
        blockTicketNavigation(deleteBtn);
        blockTicketNavigation(cancelBtn);
        blockTicketNavigation(saveBtn);

        saveBtn.addEventListener("click", e => {
            e.preventDefault();
            e.stopPropagation();

            const value = textarea.value.trim();

            if (value) {
                state.notes[ticketId] = value;
            } else {
                delete state.notes[ticketId];
            }

            save();
            closeAllNotePopups();
            apply();
        });

        deleteBtn.addEventListener("click", e => {
            e.preventDefault();
            e.stopPropagation();

            delete state.notes[ticketId];

            save();
            closeAllNotePopups();
            apply();
        });

        cancelBtn.addEventListener("click", e => {
            e.preventDefault();
            e.stopPropagation();

            closeAllNotePopups();
        });

        document.body.appendChild(popup);

        window.setTimeout(() => textarea.focus(), 30);
    }

    document.addEventListener("click", e => {
        if (!e.target.closest(".tm-note-popup") && !e.target.closest(".tm-note-btn")) {
            closeAllNotePopups();
        }
    });

    function decorateCaseCell(row, indexes) {
        const cell = row.querySelectorAll("td")[indexes.case];
        const ticketId = getTicketId(row, indexes);

        if (!cell || !ticketId || cell.querySelector(".tm-cat-ui")) return;

        const ui = document.createElement("span");
        ui.className = "tm-cat-ui";

        const badge = document.createElement("span");
        badge.className = "tm-badge";

        const select = document.createElement("select");
        select.className = "tm-category-select";

        const autoOption = document.createElement("option");
        autoOption.value = "__auto";
        autoOption.textContent = "Auto";
        select.appendChild(autoOption);

        GROUPS.forEach(group => {
            const option = document.createElement("option");
            option.value = group.id;
            option.textContent = group.title;
            select.appendChild(option);
        });

        blockTicketNavigation(ui);
        blockTicketNavigation(select);

        select.addEventListener("change", e => {
            e.preventDefault();
            e.stopPropagation();

            setManualCategory(ticketId, select.value);

            window.setTimeout(apply, 50);
        });

        ui.appendChild(badge);
        ui.appendChild(select);
        cell.appendChild(ui);

        const noteWrap = document.createElement("span");
        noteWrap.className = "tm-note-wrap";

        const noteBtn = document.createElement("button");
        noteBtn.type = "button";
        noteBtn.className = "tm-note-btn";
        noteBtn.textContent = "📝";
        noteBtn.title = "Lokale Notiz";

        blockTicketNavigation(noteWrap);
        blockTicketNavigation(noteBtn);

        noteBtn.addEventListener("click", e => {
            e.preventDefault();
            e.stopPropagation();

            if (openNoteTicketId === ticketId) {
                closeAllNotePopups();
            } else {
                openNotePopup(noteBtn, ticketId);
            }
        });

        noteWrap.appendChild(noteBtn);
        cell.appendChild(noteWrap);
    }

    function decorateAgeCell(row, indexes) {
        const ageCell = row.querySelectorAll("td")[indexes.age];
        if (!ageCell) return;

        ageCell.classList.remove("tm-age-warn", "tm-age-hot", "tm-age-old");

        const age = getAgeMinutes(row, indexes);

        if (age >= 100 * 1440) {
            ageCell.classList.add("tm-age-old");
        } else if (age >= 30 * 1440) {
            ageCell.classList.add("tm-age-hot");
        } else if (age >= 7 * 1440) {
            ageCell.classList.add("tm-age-warn");
        }
    }

    function removeOldHeaders(tbody) {
        tbody.querySelectorAll(".tm-group-header").forEach(row => row.remove());
    }

    function apply() {
        const table = findTable();
        if (!table) return;

        injectStyles();
        //buildToolbar(table);

        const indexes = getIndexes(table);
        const tbody = table.querySelector("tbody");
        if (!tbody) return;

        removeOldHeaders(tbody);

        const rows = [...tbody.querySelectorAll("tr")].filter(row => !row.classList.contains("tm-group-header"));

        const counts = {};
        GROUPS.forEach(g => counts[g.id] = 0);

        rows.forEach(row => {
            decorateCaseCell(row, indexes);
            decorateAgeCell(row, indexes);

            const ticketId = getTicketId(row, indexes);
            const caseCell = row.querySelectorAll("td")[indexes.case];

            const effectiveCategory = getEffectiveCategory(row, indexes);
            const manualCategory = getManualCategory(ticketId);
            const group = GROUPS.find(g => g.id === effectiveCategory) || GROUPS[0];

            counts[group.id] = (counts[group.id] || 0) + 1;

            const badge = caseCell.querySelector(".tm-badge");
            const select = caseCell.querySelector(".tm-category-select");
            const noteBtn = caseCell.querySelector(".tm-note-btn");

            caseCell.style.background = group.color || "";
            caseCell.style.fontWeight = group.id ? "bold" : "";

            if (badge) {
    const marker = manualCategory === null
        ? " <span class='tm-auto-marker'>A</span>"
        : "";

    badge.innerHTML = group.id ? `${group.short}${marker}` : `Keine${marker}`;
    badge.style.background = group.color || "#eee";
    badge.title = manualCategory === null
        ? "Automatisch erkannt"
        : "Manuell gesetzt";
}

            if (select) {
    select.value = manualCategory === null ? "__auto" : manualCategory;
    select.title = manualCategory === null ? "Auto-Erkennung aktiv" : "Manuell gesetzt";
}

            if (noteBtn) {
                const note = state.notes[ticketId] || "";
                noteBtn.classList.toggle("has-note", !!note);
                noteBtn.title = note ? `Notiz: ${note}` : "Lokale Notiz";
            }

            row.classList.toggle("tm-row-starred", isStarred(row, indexes));

            const visible =
                activeFilter === "all" ||
                (activeFilter === "none" && !effectiveCategory) ||
                activeFilter === effectiveCategory;

            row.style.display = visible ? "" : "none";
        });

        updateToolbarCounts(counts);

        rows.sort((a, b) => {
            const aCategory = getEffectiveCategory(a, indexes);
            const bCategory = getEffectiveCategory(b, indexes);

            const aGroup = GROUPS.find(g => g.id === aCategory) || GROUPS[0];
            const bGroup = GROUPS.find(g => g.id === bCategory) || GROUPS[0];

            if (aGroup.order !== bGroup.order) return aGroup.order - bGroup.order;

            return getAgeMinutes(a, indexes) - getAgeMinutes(b, indexes);
        });

        let lastGroup = null;

        rows.forEach(row => {
            const category = getEffectiveCategory(row, indexes);
            const group = GROUPS.find(g => g.id === category) || GROUPS[0];

            const visible =
                activeFilter === "all" ||
                (activeFilter === "none" && !category) ||
                activeFilter === category;

            if (visible && group.id !== lastGroup) {
                const header = document.createElement("tr");
                header.className = "tm-group-header";

                const td = document.createElement("td");
                td.colSpan = row.children.length;
                td.textContent = `━━ ${group.title} (${counts[group.id] || 0}) ━━`;
                header.appendChild(td);

                tbody.appendChild(header);
                lastGroup = group.id;
            }

            tbody.appendChild(row);
        });
    }

function simplifyTicketSearch() {
    const interval = setInterval(() => {
        const attr = document.querySelector("#Attribute");
        if (!attr) return;

        attr.innerHTML =
            '<option value="TicketNumber">Ticket-Nummer</option>';

        attr.value = "TicketNumber";
        attr.dispatchEvent(
            new Event("change", { bubbles: true })
        );

        clearInterval(interval);
    }, 300);
}

setTimeout(() => {
    apply();
    simplifyTicketSearch();
}, 1000);

})();
