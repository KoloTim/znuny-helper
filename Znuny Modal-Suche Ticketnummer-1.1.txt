// ==UserScript==
// @name         Znuny Modal-Suche Ticketnummer
// @namespace    hsrw-modal-search-ticketnumber
// @version      1.1
// @match        https://otrs.staff.hsrw/otrs/index.pl*
// @grant        none
// ==/UserScript==

(function () {
    "use strict";

    function fixSearchModal() {
        const form = document.querySelector("#SearchForm");
        const attr = document.querySelector("#Attribute");
        if (!form || !attr) return;

        if (attr.dataset.tmFixed === "1") return;
        attr.dataset.tmFixed = "1";

        // Ticket-Nummer fest auswählen
        attr.value = "TicketNumber";
        attr.dispatchEvent(new Event("change", { bubbles: true }));

        // Moderne Fake-Dropdown-Anzeige auch setzen, falls Znuny sie nutzt
        const visibleInput =
            attr.nextElementSibling?.querySelector("input") ||
            attr.parentElement?.querySelector("input");

        if (visibleInput) {
            visibleInput.value = "Ticket-Nummer";
        }

        // Nur Ticket-Nummer im Dropdown lassen
        [...attr.options].forEach(option => {
            if (option.value !== "TicketNumber") {
                option.remove();
            }
        });

        console.log("[TM] Znuny Suche: Ticket-Nummer gesetzt");
    }

    const observer = new MutationObserver(fixSearchModal);

    observer.observe(document.body, {
        childList: true,
        subtree: true
    });

    setInterval(fixSearchModal, 500);
})();
