// ==UserScript==
// @name         Znuny/OTRS Attachment Preview Overlay
// @namespace    local-otrs-attachments
// @version      1.5
// @description  Zeigt PDF- und Bild-Anhänge direkt als großes Overlay per Blob-Vorschau
// @match        https://otrs.staff.hsrw/otrs/index.pl*
// @grant        none
// ==/UserScript==

(function () {
    "use strict";

    function injectStyles() {

    function getExtension(text) {
        const clean = decodeURIComponent(String(text).split("?")[0].split("#")[0]).toLowerCase();
        const match = clean.match(/\.([a-z0-9]+)$/);
        return match ? match[1] : "";
    }

    function getFileName(a, index) {
        const text = a.innerText.trim();
        if (text) return text.split("\n")[0].trim();

        const title = a.getAttribute("title");
        if (title) return title.trim();

        return `Anhang-${index + 1}`;
    }

    function looksLikeAttachmentLink(a) {
        const href = (a.getAttribute("href") || "").toLowerCase();

        // Wichtig:
        // Nicht mehr über Text ".pdf" erkennen, weil sonst der Artikel-/Ticket-Titel erwischt wird.
        // Nur echte Znuny/OTRS Attachment-Download-URLs.
        return (
            href.includes("action=agentticketattachment") ||
            href.includes("subaction=download") ||
            href.includes("subaction=attachment") ||
            href.includes("downloadattachment") ||
            href.includes("attachmentid=")
        );
    }

    function guessType(fileName, blobType) {
        const ext = getExtension(fileName);

        if (ext === "pdf" || blobType.includes("pdf")) return "pdf";

        if (
            ["png", "jpg", "jpeg", "gif", "webp", "bmp"].includes(ext) ||
            blobType.startsWith("image/")
        ) {
            return "image";
        }

        return "unknown";
    }

    function stop(e) {
        e.preventDefault();
        e.stopPropagation();
    }

    function closePreview() {
        document.querySelectorAll(".tm-preview-backdrop").forEach(el => el.remove());
    }

    async function openPreview(a, fileName) {
        closePreview();

        const backdrop = document.createElement("div");
        backdrop.className = "tm-preview-backdrop";

        const modal = document.createElement("div");
        modal.className = "tm-preview-modal";

        const header = document.createElement("div");
        header.className = "tm-preview-header";

        const title = document.createElement("div");
        title.className = "tm-preview-title";
        title.textContent = fileName;

        const actions = document.createElement("div");
        actions.className = "tm-preview-actions";

        const openTab = document.createElement("a");
        openTab.href = a.href;
        openTab.target = "_blank";
        openTab.rel = "noopener";
        openTab.textContent = "↗ Tab";

        const closeBtn = document.createElement("button");
        closeBtn.type = "button";
        closeBtn.textContent = "Schließen";

        actions.appendChild(openTab);
        actions.appendChild(closeBtn);

        header.appendChild(title);
        header.appendChild(actions);

        const content = document.createElement("div");
        content.className = "tm-preview-content";
        content.innerHTML = `<div class="tm-preview-loading">Lade Vorschau...</div>`;

        modal.appendChild(header);
        modal.appendChild(content);
        backdrop.appendChild(modal);
        document.body.appendChild(backdrop);

        closeBtn.onclick = e => {
            stop(e);
            closePreview();
        };

        backdrop.addEventListener("click", e => {
            if (e.target === backdrop) closePreview();
        });

        document.addEventListener("keydown", function escHandler(e) {
            if (e.key === "Escape") {
                closePreview();
                document.removeEventListener("keydown", escHandler);
            }
        });

        try {
            const response = await fetch(a.href, {
                credentials: "include",
                cache: "no-store"
            });

            if (!response.ok) {
                throw new Error(`HTTP ${response.status}`);
            }

            const blob = await response.blob();
            const blobUrl = URL.createObjectURL(blob);
            const type = guessType(fileName, blob.type);

            content.innerHTML = "";

            if (type === "pdf") {
                const iframe = document.createElement("iframe");
                iframe.src = blobUrl;
                content.appendChild(iframe);
            } else if (type === "image") {
                const img = document.createElement("img");
                img.src = blobUrl;
                content.appendChild(img);
            } else {
                content.innerHTML = `
                    <div class="tm-preview-loading">
                        Für diesen Dateityp ist keine direkte Vorschau möglich.<br><br>
                        <a href="${blobUrl}" target="_blank" rel="noopener">Datei öffnen</a>
                    </div>
                `;
            }

            openTab.href = blobUrl;

        } catch (err) {
            content.innerHTML = `
                <div class="tm-preview-loading">
                    Vorschau konnte nicht geladen werden:<br>
                    ${err.message}
                </div>
            `;
        }
    }

    function findAttachmentContainer(a) {
        let el = a.parentElement;

        for (let i = 0; i < 6 && el; i++) {
            const text = el.innerText || "";
            const hasSize = /\b\d+[,.]?\d*\s*(KB|MB|GB)\b/i.test(text);
            const hasOnlySmallAttachmentArea = text.length < 500;

            if (hasSize && hasOnlySmallAttachmentArea) {
                return el;
            }

            el = el.parentElement;
        }

        return a.parentElement;
    }

    function makeButton() {
        const btn = document.createElement("button");
        btn.type = "button";
        btn.className = "tm-attachment-btn";
        btn.innerHTML = "👁️ Vorschau";
        btn.title = "Anhang groß anzeigen";
        return btn;
    }

    function enhanceAttachments() {
        injectStyles();

        const links = [...document.querySelectorAll("a[href]")].filter(looksLikeAttachmentLink);

        links.forEach((a, index) => {
            if (a.dataset.tmAttachmentEnhanced === "1") return;
            a.dataset.tmAttachmentEnhanced = "1";

            const fileName = getFileName(a, index);
            const container = findAttachmentContainer(a);

            if (!container) return;

            container.classList.add("tm-attachment-container");

            const previewBtn = makeButton();

            previewBtn.onclick = e => {
                stop(e);
                openPreview(a, fileName);
            };

            container.appendChild(previewBtn);
        });
    }

    function init() {
        enhanceAttachments();

        const observer = new MutationObserver(() => {
            enhanceAttachments();
        });

        observer.observe(document.body, {
            childList: true,
            subtree: true
        });
    }

    setTimeout(init, 1000);
})();
